cmake_minimum_required(VERSION 3.22)
project(flva LANGUAGES C CXX)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
set(NATIVE ${ROOT}/../native)
set(SHERPA ${ROOT}/src/main/jniLibs/${ANDROID_ABI}/libsherpa-onnx-c-api.so)
if(NOT EXISTS ${SHERPA})
  message(FATAL_ERROR "Provision pinned sherpa v1.12.14 libraries with tools/provision_runtime.py; no runtime download is performed.")
endif()
add_library(sherpa SHARED IMPORTED)
set_target_properties(sherpa PROPERTIES IMPORTED_LOCATION ${SHERPA})
file(GLOB CORE ${NATIVE}/src/*.cpp)
add_library(flva SHARED bridge.cpp ${CORE})
target_include_directories(flva PRIVATE ${NATIVE}/include ${NATIVE}/src)
target_compile_features(flva PRIVATE cxx_std_17)
target_link_libraries(flva sherpa android log)

option(FLVA_ENABLE_LOCAL_LLM "Build optional local GGUF intelligence" OFF)
if(FLVA_ENABLE_LOCAL_LLM)
  set(FLVA_BUILD_LLM_SMOKE OFF CACHE BOOL "" FORCE)
  add_subdirectory(${NATIVE}/llm ${CMAKE_CURRENT_BINARY_DIR}/llm)
  target_compile_definitions(flva PRIVATE FLVA_ENABLE_LLM=1)
  target_link_libraries(flva flva_llm_adapter)
endif()
